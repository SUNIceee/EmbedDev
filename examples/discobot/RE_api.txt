10. CircArray -- circular array
================================================================================

typedef struct CircArray {
    char *buf;
    uint32_t size;
    bool enabled;
    uint32_t n_r;
    uint32_t n_w;
} CircArray;

void initCircArray(CircArray *arr, int size);
  // calloc(size,1). Set size/enabled=1/n_r=0/n_w=0. Already enabled -> print error and skip.

int buf_putbyte(CircArray *arr, char c);
  // !full→arr->buf[n_w%size]=c→n_w++→ret1。full→ret0。

char buf_getbyte(CircArray *arr);
  // !empty→ret arr->buf[n_r%size]→n_r++。empty→ret0。

bool buf_empty(CircArray *arr);
  // n_r==n_w

bool buf_full(CircArray *arr);
  // !empty && (n_r%size)==(n_w%size)

int buf_available(CircArray *arr);
  // n_w - n_r

bool buf_resize(CircArray *arr, int newSize);
  // realloc(arr->buf,newSize). Failure -> false. Success -> size=newSize -> true.

bool buf_delete(CircArray *arr);
  // buf_clear -> free(arr->buf) -> zero fields -> enabled=false. arr==NULL -> false.

void buf_clear(CircArray *arr);
  // memset all bytes to 0. Do not change n_r/n_w/size/enabled.

================================================================================
13. Host testing and platform isolation
================================================================================

Do not unconditionally include SPL headers such as stm32f4xx.h / stm32f4xx_gpio.h / misc.h.
Guard platform code with DISCOBOT_TARGET; otherwise use standard C/C++ host stubs and simulated registers.
All pure-algorithm APIs must link and run on the host.
Provide no-op stubs for SPL configuration calls (RCC_*ClockCmd / GPIO_Init / USART_Init, etc.) in host mode.
